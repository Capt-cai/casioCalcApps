cmake_minimum_required(VERSION 3.15)
project(ECEN21 C)

include(GenerateG1A)
include(GenerateG3A)
include(Fxconv)
find_package(Gint 2.9 REQUIRED)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../common/common.cmake)

option(ECEN21_TEST_BUILD "Build the target controller test harness" OFF)
option(ECEN21_BENCH_BUILD "Build the maximum-domain measurement harness" OFF)
set(CORE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../experiments/boolean)
if(ECEN21_BENCH_BUILD)
  set(APP_TARGET EC21Bench)
  set(APP_MAIN tests/hdl_benchmark_main.c)
elseif(ECEN21_TEST_BUILD)
  set(APP_TARGET EC21Test)
  set(APP_MAIN tests/target_main.c tests/controller_checks.c tests/solver_checks.c tests/numeric_checks.c tests/flipflop_checks.c tests/sequential_checks.c tests/connection_checks.c tests/timing_checks.c tests/timing_sample_checks.c tests/combinational_checks.c tests/hdl_checks.c src/controller.c)
else()
  set(APP_TARGET ECEN21)
  set(APP_MAIN src/main.c src/boolean_ui.c src/numeric_ui.c src/flipflop_ui.c src/sequential_ui.c src/timing_ui.c src/combinational_ui.c src/hdl_ui.c src/hdl_storage.c)
endif()
casio_app(${APP_TARGET} SOURCES
  ${APP_MAIN} src/solver.c src/numeric.c src/flipflop.c src/sequential.c src/seq_connections.c src/timing.c src/timing_samples.c src/combinational.c src/hdl.c src/hdl_export.c ${CORE_DIR}/exact_boolean.c)
target_include_directories(${APP_TARGET} PRIVATE src ${CORE_DIR})
target_compile_options(${APP_TARGET} PRIVATE -Werror -fstack-usage)
